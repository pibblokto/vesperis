# Vesperis — CMake build (the Makefile remains the quick path on macOS).
#   cmake -B build-cmake && cmake --build build-cmake -j
#   cmake -B build-cmake -DVESPERIS_BUILD_GAME=OFF        # headless harness only, no raylib needed
#   cmake -B build-cmake -DVESPERIS_FETCH_RAYLIB=ON       # download and build raylib 5.5 when none is installed (Linux, Windows)
#   emcmake cmake -B build-web -DPLATFORM=Web ...       # web build: see web/build_web.sh (M6-04)
cmake_minimum_required(VERSION 3.16)
project(vesperis LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(VESPERIS_BUILD_GAME "Build the windowed game (needs raylib 4.5+ / 5.x)" ON)
option(VESPERIS_FETCH_RAYLIB "Fetch and build raylib with FetchContent when it is not installed" OFF)
option(VESPERIS_WERROR "Treat warnings as errors in core/ and galaxy/ (M7-06)" ON)

file(GLOB CORE_SOURCES CONFIGURE_DEPENDS
     src/core/*.cpp src/galaxy/*.cpp src/space/*.cpp src/surface/*.cpp src/game/*.cpp)
add_library(vesperis_core STATIC ${CORE_SOURCES})
target_include_directories(vesperis_core PUBLIC src)
find_package(Threads REQUIRED)
target_link_libraries(vesperis_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(vesperis_core PRIVATE /W3 /wd4244 /wd4267 /wd4305 /utf-8 /EHsc)
  target_compile_definitions(vesperis_core PUBLIC _USE_MATH_DEFINES NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(vesperis_core PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
  if(VESPERIS_WERROR)
    # the deterministic layers must build clean: warnings are errors there (M7-06)
    file(GLOB STRICT_SOURCES CONFIGURE_DEPENDS src/core/*.cpp src/galaxy/*.cpp)
    set_source_files_properties(${STRICT_SOURCES} PROPERTIES COMPILE_OPTIONS "-Werror")
  endif()
endif()

add_executable(vesperis_test src/main_headless.cpp)
target_link_libraries(vesperis_test PRIVATE vesperis_core)
if(MSVC)
  target_compile_definitions(vesperis_test PRIVATE _USE_MATH_DEFINES NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()

if(VESPERIS_BUILD_GAME)
  set(VESPERIS_RAYLIB_TARGET "")
  if(NOT VESPERIS_FETCH_RAYLIB)
    find_package(raylib QUIET)
    if(raylib_FOUND)
      set(VESPERIS_RAYLIB_TARGET raylib)
    else()
      find_package(PkgConfig QUIET)
      if(PkgConfig_FOUND)
        pkg_check_modules(RAYLIB IMPORTED_TARGET raylib)
        if(RAYLIB_FOUND)
          set(VESPERIS_RAYLIB_TARGET PkgConfig::RAYLIB)
        endif()
      endif()
    endif()
  endif()
  if(VESPERIS_RAYLIB_TARGET STREQUAL "" AND (VESPERIS_FETCH_RAYLIB OR NOT APPLE))
    # M6-03: no system raylib: fetch 5.5 and build it as a static library (needs the X11/Wayland dev packages on Linux)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib
      GIT_REPOSITORY https://github.com/raysan5/raylib.git
      GIT_TAG 5.5
      GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(raylib)
    set(VESPERIS_RAYLIB_TARGET raylib)
  endif()
  if(NOT VESPERIS_RAYLIB_TARGET STREQUAL "")
    add_executable(vesperis src/main_raylib.cpp src/platform_raylib.cpp)
    target_link_libraries(vesperis PRIVATE vesperis_core ${VESPERIS_RAYLIB_TARGET})
    if(MSVC)
      target_compile_definitions(vesperis PRIVATE _USE_MATH_DEFINES NOMINMAX _CRT_SECURE_NO_WARNINGS)
    endif()
    if(APPLE)
      target_link_libraries(vesperis PRIVATE "-framework CoreVideo" "-framework IOKit" "-framework Cocoa" "-framework OpenGL")
    endif()
    if(EMSCRIPTEN)
      # M6-04: the browser build; the shell page and the asset rules live in web/
      set_target_properties(vesperis PROPERTIES SUFFIX ".html")
      target_link_options(vesperis PRIVATE -sUSE_GLFW=3 -sASYNCIFY -sALLOW_MEMORY_GROWTH=1 -sTOTAL_STACK=4MB -sPTHREAD_POOL_SIZE=4 -pthread
                          --shell-file ${CMAKE_SOURCE_DIR}/web/shell.html -sFORCE_FILESYSTEM=1 -lidbfs.js)
      target_compile_options(vesperis PRIVATE -pthread)
      target_compile_options(vesperis_core PRIVATE -pthread)
    endif()
    install(TARGETS vesperis RUNTIME DESTINATION .)
    install(FILES README.md DESTINATION .)
    install(DIRECTORY docs DESTINATION .)
  else()
    message(WARNING "raylib not found: building only vesperis_test (install raylib, pass -DVESPERIS_FETCH_RAYLIB=ON, or -DVESPERIS_BUILD_GAME=OFF)")
  endif()
endif()

# Headless checks; run from the source tree so shots/ and tests/ are found.
enable_testing()
foreach(mode flow regress settings landmaps fuzz input seasons audio unit)
  add_test(NAME ${mode} COMMAND vesperis_test ${mode} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endforeach()
add_test(NAME bench COMMAND vesperis_test bench check WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})

# M6-03 packaging: `cpack -B build-cmake` after a build produces a zip (a drag-and-drop bundle on macOS via packaging/make_app.sh)
set(CPACK_PACKAGE_NAME "vesperis")
set(CPACK_PACKAGE_VERSION "1.2.0")
set(CPACK_GENERATOR "ZIP")
include(CPack)
